WITH wk AS (SELECT week_end_date AS we, DATE_SUB(week_end_date, INTERVAL 90 DAY) AS ws FROM `petcircle-warehouse-cloud.business_layer.reference__calendar_4_4_5_week` WHERE is_latest_merch_complete_week = TRUE), r AS (SELECT DATE_TRUNC(v.review_date, WEEK(MONDAY)) AS week_start, v.review_id, v.net_promoter_count AS np, v.review_count_promoter AS pr, v.review_count_neutral AS pa, v.review_count_detractor AS de, v.review_text, v.customer_review_order_mapping[SAFE_OFFSET(0)].order_id AS oid FROM `petcircle-warehouse-cloud.business_layer.customer__customer_review` v CROSS JOIN wk WHERE v.review_date BETWEEN wk.ws AND wk.we) SELECT SUBSTR(TO_HEX(SHA256(CAST(r.review_id AS STRING))), 1, 16) AS rid, CAST(r.week_start AS STRING) AS week_start, CASE WHEN r.pr = 1 THEN 'Promoter' WHEN r.pa = 1 THEN 'Passive' ELSE 'Detractor' END AS band, r.review_text AS comment_text FROM r WHERE LENGTH(TRIM(r.review_text)) >= 3 ORDER BY rid
